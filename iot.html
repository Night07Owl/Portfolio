// ARDUINO C++ SOURCE & SYSTEM TELEMETRY
task1esp32.ino 
#include <WiFi.h>
#include <WebServer.h>

const char* ssid     = "protosem";
const char* password = "Proto#123";

#define LED_PIN 2

WebServer server(80);

void handleRoot()
{
  String html = R"rawliteral(
<!DOCTYPE html>
<html>
<head>
  <title>ESP32 LED Control</title>
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <style>
    body { font-family: Arial; text-align: center; margin-top: 80px; }
    h1 { font-size: 35px; }
    button { width: 150px; height: 60px; font-size: 22px; margin: 15px; cursor: pointer; }
  </style>
</head>
<body>
  <h1>ESP32 LED Control</h1>
  <a href="/on"><button>ON</button></a>
  <a href="/off"><button>OFF</button></a>
</body>
</html>
)rawliteral";

  server.send(200, "text/html", html);
}

void handleLEDOn()
{
  digitalWrite(LED_PIN, HIGH);
  server.send(200, "text/html", "<h1>LED is ON</h1><a href='/'>Back</a>");
}

void handleLEDOff()
{
  digitalWrite(LED_PIN, LOW);
  server.send(200, "text/html", "<h1>LED is OFF</h1><a href='/'>Back</a>");
}

void setup()
{
  Serial.begin(115200);

  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);

  // Connect to Wi-Fi
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");

  while (WiFi.status() != WL_CONNECTED)
  {
    delay(500);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("WiFi connected!");

  Serial.print("ESP32 IP address: ");
  Serial.println(WiFi.localIP());

  // Web server routes
  server.on("/", handleRoot);
  server.on("/on", handleLEDOn);
  server.on("/off", handleLEDOff);

  server.begin();
  Serial.println("Web server started!");
}

void loop()
{
  server.handleClient();
}

  }
  http.end();
  lastTrigger = millis();
}

Objective

PROJECT GOALS
  • To develop an end-to-end autonomous edge-controlled IoT system integrating environmental sensors, electromechanical actuators, and cloud synchronization.
  • To implement software hysteresis with a calibrated 300-point deadband (1200 / 1500 raw ADC) to eliminate high-frequency relay contact chatter and destructive arcing.
  • To architect a dual-mode operational controller permitting seamless runtime switching between autonomous edge control and remote manual cloud override.
  • To build a client-side streaming CSV telemetry generation engine directly in the web interface for offline engineering audits and historical data analysis without server dependencies.
02

Problem Being Addressed

PAIN POINTS & FAILURES
  • Single-Threshold Relay Chatter & Mechanical Failure: When a system toggles based on a single threshold (e.g. Light < 1500), sensor noise causes rapid oscillation between 1499 and 1501, making mechanical relays switch 10+ times per second, generating destructive electrical arcs.
  • Cloud Dependency Single Point of Failure: Standard IoT architectures offload automation logic to cloud brokers; when internet connectivity drops or cloud APIs experience latency, local actuators fail to respond to critical environmental changes.
  • Telemetry Data Lock-In & Inaccessible Audits: Microcontroller memory and cloud JSON trees cannot be easily exported for offline analysis by engineers using spreadsheet tools (Excel, pandas, MATLAB) without heavy backend server code.
03

Project Summary

HIGH-LEVEL OVERVIEW
  • The ESP32 continuously samples an LDR photoresistor on GPIO 34 through internal 12-bit ADC conversion alongside a calibrated DHT11 climate sensor on GPIO 4.
  • An embedded software hysteresis algorithm executes locally at the edge: the 5V relay energizes only when light plunges below 1200 ADC and de-energizes only when light rises above 1500 ADC, completely ignoring the 300-point intermediate zone.
  • Real-time operating metrics (mode, light level, temperature, humidity, bulb status) stream continuously to Google Firebase Realtime Database over encrypted WebSockets.
  • The web dashboard incorporates an instant client-side RFC 4180 CSV export generator that serializes recorded telemetry in browser memory into downloadable spreadsheet files on demand.

#define WIFI_PASS     "YOUR_WIFI_PASSWORD"
#define FIREBASE_HOST "forge-iot-telemetry-default-rtdb.firebaseio.com"
#define FIREBASE_AUTH "YOUR_DATABASE_SECRET"

#define DHTPIN 4
#define LDRPIN 34
#define RELAYPIN 18

// Software Hysteresis Thresholds (300-point anti-chatter deadband)
const int THRESHOLD_DARK  = 1200;
const int THRESHOLD_LIGHT = 1500;

DHT dht(DHTPIN, DHT11);
FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

unsigned long lastLogTime = 0;
bool currentBulbState = false;
String currentMode = "AUTOMATIC";

void setup() {
  Serial.begin(115200);
  pinMode(RELAYPIN, OUTPUT);
  digitalWrite(RELAYPIN, LOW);
  dht.begin();

  WiFi.begin(WIFI_SSID, WIFI_PASS);
  while (WiFi.status() != WL_CONNECTED) { delay(500); }
  
  configTime(19800, 0, "pool.ntp.org"); // UTC + 5:30 (IST)

  config.host = FIREBASE_HOST;
  config.signer.tokens.legacy_token = FIREBASE_AUTH;
  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
}

void loop() {
  float t = dht.readTemperature();
  float h = dht.readHumidity();
  int ldr = analogRead(LDRPIN);

  // 1. Retrieve Current Mode from Firebase
  if (Firebase.getString(fbdo, "/iot_system/system_state/mode")) {
    currentMode = fbdo.stringData();
  }

  // 2. Execute Hysteresis Control Logic
  if (currentMode == "AUTOMATIC") {
    if (ldr < THRESHOLD_DARK && !currentBulbState) {
      currentBulbState = true;
      digitalWrite(RELAYPIN, HIGH);
      Firebase.setBool(fbdo, "/iot_system/system_state/bulb_state", true);
    } else if (ldr > THRESHOLD_LIGHT && currentBulbState) {
      currentBulbState = false;
      digitalWrite(RELAYPIN, LOW);
      Firebase.setBool(fbdo, "/iot_system/system_state/bulb_state", false);
    }
  } else { // MANUAL OVERRIDE MODE
    if (Firebase.getBool(fbdo, "/iot_system/system_state/bulb_state")) {
      currentBulbState = fbdo.boolData();
      digitalWrite(RELAYPIN, currentBulbState ? HIGH : LOW);
    }
  }

  // 3. Periodic Historical Logging (Every 30 seconds)
  if (millis() - lastLogTime > 30000) {
    lastLogTime = millis();
    FirebaseJson json;
    json.set("temperature", t);
    json.set("humidity", h);
    json.set("light_level", ldr);
    json.set("bulb_state", currentBulbState ? "ON" : "OFF");
    json.set("mode", currentMode);
    
    Firebase.pushJSON(fbdo, "/iot_system/historical_logs", json);
  }
}

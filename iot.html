// ARDUINO C++ SOURCE & SYSTEM TELEMETRY
task1esp32.ino 
#include <WiFi.h>
#include <WebServer.h>

const char* ssid     = "protosem";
const char* password = "Proto#123";

#define LED_PIN 2

WebServer server(80);

void handleRoot()
{
  String html = R"rawliteral(
<!DOCTYPE html>
<html>
<head>
  <title>ESP32 LED Control</title>
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <style>
    body { font-family: Arial; text-align: center; margin-top: 80px; }
    h1 { font-size: 35px; }
    button { width: 150px; height: 60px; font-size: 22px; margin: 15px; cursor: pointer; }
  </style>
</head>
<body>
  <h1>ESP32 LED Control</h1>
  <a href="/on"><button>ON</button></a>
  <a href="/off"><button>OFF</button></a>
</body>
</html>
)rawliteral";

  server.send(200, "text/html", html);
}

void handleLEDOn()
{
  digitalWrite(LED_PIN, HIGH);
  server.send(200, "text/html", "<h1>LED is ON</h1><a href='/'>Back</a>");
}

void handleLEDOff()
{
  digitalWrite(LED_PIN, LOW);
  server.send(200, "text/html", "<h1>LED is OFF</h1><a href='/'>Back</a>");
}

void setup()
{
  Serial.begin(115200);

  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);

  // Connect to Wi-Fi
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");

  while (WiFi.status() != WL_CONNECTED)
  {
    delay(500);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("WiFi connected!");

  Serial.print("ESP32 IP address: ");
  Serial.println(WiFi.localIP());

  // Web server routes
  server.on("/", handleRoot);
  server.on("/on", handleLEDOn);
  server.on("/off", handleLEDOff);

  server.begin();
  Serial.println("Web server started!");
}

void loop()
{
  server.handleClient();
}

Problem 1 — Active-LOW Relay Inversion Logic

Issue: Initially, writing digitalWrite(RELAY_PIN, HIGH) when receiving the "ON" MQTT payload kept the relay de-energized, while receiving "OFF" turned the bulb ON. The optocoupled relay circuit operates on inverted active-LOW switching logic rather than active-HIGH logic.

Problem 2 — Inactive MQTT TCP Socket Timeout & Drops

Issue: During idle periods with no toggle clicks from the dashboard, the cloud MQTT broker closed the inactive TCP socket due to keep-alive expiry. The ESP32 missed subsequent toggle packets until power was manually cycled.

// TECHNICAL FIXES & HARDENING
✅

Solution 1 — Firmware Inverted Logic Calibration

Mapped the received "ON" payload to digitalWrite(RELAY_PIN, LOW) (driving 0V to complete the optocoupler circuit) and "OFF" to digitalWrite(RELAY_PIN, HIGH) (driving 3.3V to open the relay contact). This aligned firmware logic with the physical active-LOW relay module.

✅

Solution 2 — Self-Healing Non-Blocking Auto-Reconnect Loop

Implemented an automated watchdog check at the start of loop(): if (!mqtt.connected()) { connectMQTT(); }. When a socket drop occurs, the ESP32 automatically logs the disconnection and reconnects to Adafruit IO within 5 seconds without requiring a manual hardware restart.

// SYSTEM LIMITATIONS SOLVED
  • Local Subnet Confinement: Standalone web servers running directly on microcontrollers require users to be connected to the exact same local Wi-Fi router, preventing remote out-of-home access.
  • Absence of Access Control: Traditional basic IoT boards lack identity management, allowing any device on the network to switch high-voltage loads without verification.

Hardware Sensors, Circuit & Appliance Actuation

// CYBER-PHYSICAL INTERFACE
🌡️

DHT11 Sensor Connection

  • Pins Connected: VCC to 3V3, GND to GND, DATA to GPIO 4.
  • Protocol: Single-bus proprietary digital timing protocol sending 40 bits of serial data (16-bit humidity, 16-bit temperature, 8-bit checksum).
  • Purpose: Measures ambient temperature (°C) and relative humidity (%). Verified displays: 22.2°C and 49% RH.
☀️

LDR Sensor & Voltage Divider

  • Pins Connected: Leg 1 to 3V3, Leg 2 to GPIO 34 (ADC1 Channel 6) with a 10kΩ pull-down resistor to GND.
  • ADC Measurement: 12-bit resolution yielding values from 0 (maximum brightness) to 4095 (pitch dark).
  • Purpose: Provides ambient light detection. Compared against ldrThreshold: when light drops (LDR >= threshold), bulb trips ON automatically.
💡

5V Relay & 230V AC Bulb Control

  • Pins Connected: VCC to VIN (5V), GND to GND, IN to GPIO 26.
  • Logic: Active-LOW optocoupled input (LOW = Relay Energized / ON, HIGH = Relay De-energized / OFF).
  • Load Circuit: 230V AC mains lamp wired in series through Common (COM) and Normally Open (NO) screw terminals, safely isolating mains voltage.

  time_t now;
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {

void pollFirebaseControl() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  String url = "https://" + String(FIREBASE_HOST) + "/control.json";
  if (strlen(FIREBASE_AUTH) > 0) {
    url += "?auth=" + String(FIREBASE_AUTH);
  }

  http.begin(client, url);
  int httpCode = http.GET();

  if (httpCode == HTTP_CODE_OK) {
    String payload = http.getString();
    StaticJsonDocument<256> doc;
    DeserializationError error = deserializeJson(doc, payload);

    if (!error) {
      if (doc.containsKey("mode")) {
        currentMode = doc["mode"].as<String>();
      }
      if (doc.containsKey("ldrThreshold")) {
        ldrThreshold = doc["ldrThreshold"].as<int>();
      }
      
      // If manual mode, follow the user's manual bulb command
      if (currentMode == "manual") {
        if (doc.containsKey("bulb")) {
          bool requestedBulb = doc["bulb"].as<bool>();
          setRelay(requestedBulb);
        }
      }
    }
  } else {
    Serial.printf("[HTTP] GET /control.json failed, code: %d\n", httpCode);
  }
  http.end();
}

void loop() {
  unsigned long now = millis();

  // 1. Read Physical Environmental Sensors
  float t = dht.readTemperature();
  float h = dht.readHumidity();
  if (!isnan(t)) temperature = t;
  if (!isnan(h)) humidity = h;

  ldrValue = analogRead(LDR_PIN);

  // 2. Poll Firebase /control (Manual/Auto mode, bulb toggle, threshold)
  if (now - lastControlPoll >= CONTROL_INTERVAL) {
    lastControlPoll = now;
    pollFirebaseControl();
  }

  // 3. Automation Engine (Automatic Mode)
  // Logic: When LDR >= threshold -> Dark -> Turn Bulb ON
  //        When LDR <  threshold -> Bright -> Turn Bulb OFF
  if (currentMode == "automatic") {
    bool shouldBulbBeOn = (ldrValue >= ldrThreshold);
    if (bulbState != shouldBulbBeOn) {
      setRelay(shouldBulbBeOn);
      Serial.printf("[AUTO] LDR: %d vs Threshold: %d -> Bulb toggled %s\n", 
                    ldrValue, ldrThreshold, shouldBulbBeOn ? "ON" : "OFF");
    }
  }

  // 4. Periodically update Firebase /current
  if (now - lastCurrentPush >= CURRENT_INTERVAL) {
    lastCurrentPush = now;
    pushCurrentValues();
  }

  // 5. Periodically log time-series entry to Firebase /history
  if (now - lastHistoryPush >= HISTORY_INTERVAL) {
    lastHistoryPush = now;
    pushHistoryLog();
  }

  delay(50); // Small loop yield for ESP32 background tasks (Wi-Fi watchdog)
}

function updateHistory(history) {

    const table = document.getElementById("historyTable");
    table.innerHTML = "";

    historyForCSV = [];

    const timestamps = [];
    const temperatures = [];
    const humidities = [];
    const ldrValues = [];

    const entries = Object.entries(history || {})
        .sort((a, b) => {
            return Number(a[1].timestamp || 0)
                - Number(b[1].timestamp || 0);
        })
        .slice(-200);

    entries.forEach(([key, item]) => {

        const rawTimestamp = Number(item.timestamp || key || 0);

        const date = rawTimestamp > 100000000000
            ? new Date(rawTimestamp)
            : new Date(rawTimestamp * 1000);

        const dateText = isNaN(date.getTime())
            ? "Unknown"
            : date.toLocaleString();

        // Table
        const row = document.createElement("tr");

        const cells = [
            dateText,
            `${item.temperature ?? ""} °C`,
            `${item.humidity ?? ""} %`,
            item.ldr ?? "",
            item.bulb ? "ON" : "OFF",
            item.mode ?? ""
        ];

        cells.forEach((value) => {
            const cell = document.createElement("td");
            cell.textContent = value;
            row.appendChild(cell);
        });

        table.appendChild(row);

        // Graph
        timestamps.push(
            isNaN(date.getTime())
                ? String(rawTimestamp)
                : date.toLocaleTimeString()
        );

        temperatures.push(Number(item.temperature ?? 0));
        humidities.push(Number(item.humidity ?? 0));
        ldrValues.push(Number(item.ldr ?? 0));

        // CSV
        historyForCSV.push({
            timestamp: dateText,
            temperature: item.temperature ?? "",
            humidity: item.humidity ?? "",
            ldr: item.ldr ?? "",
            bulb: item.bulb ? "ON" : "OFF",
            mode: item.mode ?? ""
        });
    });

    updateChart(
        timestamps,
        temperatures,
        humidities,
        ldrValues
    );
}

Automatic Lighting Logic, LDR Calibration & Database Architecture

ALGORITHM & SCHEMA

💡 Automatic Lighting Logic & Calibration Method

  • Operating Mode Check: The ESP32 evaluates the active mode set via the dashboard. In "manual" mode, ambient sensor switching is bypassed so operator buttons retain full control.
  • Darkness Detection Logic: In "automatic" mode, when the light level drops below the calibrated threshold (ldrValue >= ldrThreshold), the ESP32 drives GPIO 26 LOW, energizing the active-LOW relay and turning the bulb ON.
  • Daylight De-energization: When natural or ambient light rises above the threshold (ldrValue < ldrThreshold), GPIO 26 is driven HIGH, turning the relay OFF.
  • LDR Calibration Baseline:
    • Ambient Daylight / Indoor Lit Room: 2100–2500 ADC
    • Covered / Dark Room State: >2800 ADC (up to 4095 ADC)
    • Direct Bright Light: <1500 ADC
    • Default Calibration Threshold: 2400 ADC (calibrated for reliable dark activation)
  • Anti-Chatter Hysteresis: A software deadband of ±150 ADC prevents rapid oscillatory relay clicking caused by transitional shadows or incandescent bulb reflection.

#define DHT_PIN          4        // GPIO 4 -> DHT11 / DHT22 Data Pin

// Replace with your local Wi-Fi credentials

void setRelay(bool state) {
  bulbState = state;
  if (RELAY_ACTIVE_LOW) {
    digitalWrite(RELAY_PIN, state ? LOW : HIGH);
  } else {
    digitalWrite(RELAY_PIN, state ? HIGH : LOW);
  }
}

/* =========================================================================
   5. HELPER: GET CURRENT UNIX TIMESTAMP (MILLISECONDS)
   ========================================================================= */
unsigned long long getEpochMillis() {
  time_t now;
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    return (unsigned long long)millis(); // Fallback if NTP not yet synced
  }
  time(&now);
  return ((unsigned long long)now) * 1000ULL;
}

/* =========================================================================
   6. FIREBASE REST: FETCH CONTROL STATE (/control.json)
   ========================================================================= */
void pollFirebaseControl() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  String url = "https://" + String(FIREBASE_HOST) + "/control.json";
  if (strlen(FIREBASE_AUTH) > 0) {
    url += "?auth=" + String(FIREBASE_AUTH);
  }

  http.begin(client, url);
  int httpCode = http.GET();

  if (httpCode == HTTP_CODE_OK) {
    String payload = http.getString();
    StaticJsonDocument<256> doc;
    DeserializationError error = deserializeJson(doc, payload);

    if (!error) {
      if (doc.containsKey("mode")) {
        currentMode = doc["mode"].as<String>();
      }
      if (doc.containsKey("ldrThreshold")) {
        ldrThreshold = doc["ldrThreshold"].as<int>();
      }
      
      // If manual mode, follow the user's manual bulb command
      if (currentMode == "manual") {
        if (doc.containsKey("bulb")) {
          bool requestedBulb = doc["bulb"].as<bool>();
          setRelay(requestedBulb);
        }
      }
    }
  } else {
    Serial.printf("[HTTP] GET /control.json failed, code: %d\n", httpCode);
  }
  http.end();
}

/* =========================================================================
   7. FIREBASE REST: UPDATE LIVE STATE (/current.json)
   ========================================================================= */
void pushCurrentValues() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  String url = "https://" + String(FIREBASE_HOST) + "/current.json";
  if (strlen(FIREBASE_AUTH) > 0) {
    url += "?auth=" + String(FIREBASE_AUTH);
  }

  StaticJsonDocument<256> doc;
  doc["temperature"] = serialized(String(temperature, 1));
  doc["humidity"]    = serialized(String(humidity, 1));
  doc["ldr"]         = ldrValue;
  doc["bulb"]        = bulbState;
  doc["mode"]        = currentMode;

  String requestBody;
  serializeJson(doc, requestBody);

  http.begin(client, url);
  http.addHeader("Content-Type", "application/json");
  int httpCode = http.PUT(requestBody); // PUT updates / replaces /current

  if (httpCode != HTTP_CODE_OK) {
    Serial.printf("[HTTP] PUT /current.json failed, code: %d\n", httpCode);
  }
  http.end();
}

/* =========================================================================
   8. FIREBASE REST: LOG HISTORICAL RECORD (/history.json)
   ========================================================================= */
void pushHistoryLog() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  String url = "https://" + String(FIREBASE_HOST) + "/history.json";
  if (strlen(FIREBASE_AUTH) > 0) {
    url += "?auth=" + String(FIREBASE_AUTH);

  StaticJsonDocument<256> doc;
  doc["timestamp"]   = getEpochMillis();
  doc["temperature"] = serialized(String(temperature, 1));
  doc["humidity"]    = serialized(String(humidity, 1));
  doc["ldr"]         = ldrValue;
  doc["bulb"]        = bulbState;
  doc["mode"]        = currentMode;

  String requestBody;
  serializeJson(doc, requestBody);

  http.begin(client, url);
  http.addHeader("Content-Type", "application/json");
  int httpCode = http.POST(requestBody); // POST generates a unique push-key in /history

  if (httpCode != HTTP_CODE_OK) {
    Serial.printf("[HTTP] POST /history.json failed, code: %d\n", httpCode);
  } else {
    Serial.println("[FIREBASE] Telemetry snapshot saved to /history");

void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n=== SMART ROOM IOT INITIALIZATION ===");

  // Pin Configurations
  pinMode(RELAY_PIN, OUTPUT);
  pinMode(STATUS_LED_PIN, OUTPUT);
  pinMode(LDR_PIN, INPUT);

  // Initial actuator state: OFF
  setRelay(false);
  digitalWrite(STATUS_LED_PIN, LOW);

  // Initialize DHT Sensor
  dht.begin();

  // Insecure TLS (bypasses CA verification for simplicity & reliability on ESP32)
  client.setInsecure();

  // Wi-Fi Connection
  Serial.printf("[WIFI] Connecting to SSID: %s", WIFI_SSID);
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  while (WiFi.status() != WL_CONNECTED) {
    digitalWrite(STATUS_LED_PIN, !digitalRead(STATUS_LED_PIN)); // Blink while connecting
    delay(400);
    Serial.print(".");
  }

  digitalWrite(STATUS_LED_PIN, HIGH); // Solid ON when connected
  Serial.println("\n[WIFI] Connected successfully!");
  Serial.printf("[WIFI] Assigned ESP32 IP: %s\n", WiFi.localIP().toString().c_str());

  // Initialize NTP time for timestamps
  configTime(GMT_OFFSET_SEC, DAYLIGHT_OFFSET, NTP_SERVER);
  Serial.println("[NTP] Time synchronization service initialized.");
}

/* =========================================================================
   10. MAIN EVENT LOOP
   ========================================================================= */
void loop() {
  unsigned long now = millis();

  // 1. Read Physical Environmental Sensors
  float t = dht.readTemperature();
  float h = dht.readHumidity();
  if (!isnan(t)) temperature = t;
  if (!isnan(h)) humidity = h;

  ldrValue = analogRead(LDR_PIN);

  // 2. Poll Firebase /control (Manual/Auto mode, bulb toggle, threshold)
  if (now - lastControlPoll >= CONTROL_INTERVAL) {
    lastControlPoll = now;
    pollFirebaseControl();
  }

  // 3. Automation Engine (Automatic Mode)
  // Logic: When LDR >= threshold -> Dark -> Turn Bulb ON
  //        When LDR <  threshold -> Bright -> Turn Bulb OFF
  if (currentMode == "automatic") {
    bool shouldBulbBeOn = (ldrValue >= ldrThreshold);
    if (bulbState != shouldBulbBeOn) {
      setRelay(shouldBulbBeOn);
      Serial.printf("[AUTO] LDR: %d vs Threshold: %d -> Bulb toggled %s\n", 
                    ldrValue, ldrThreshold, shouldBulbBeOn ? "ON" : "OFF");
    }
  }

  // 4. Periodically update Firebase /current
  if (now - lastCurrentPush >= CURRENT_INTERVAL) {
    lastCurrentPush = now;
    pushCurrentValues();
  }

  // 5. Periodically log time-series entry to Firebase /history
  if (now - lastHistoryPush >= HISTORY_INTERVAL) {
    lastHistoryPush = now;
    pushHistoryLog();
  }

  delay(50); // Small loop yield for ESP32 background tasks (Wi-Fi watchdog)
}

function updateHistory(history) {

    const table = document.getElementById("historyTable");
    table.innerHTML = "";

    historyForCSV = [];

    const timestamps = [];
    const temperatures = [];
    const humidities = [];
    const ldrValues = [];

    const entries = Object.entries(history || {})
        .sort((a, b) => {
            return Number(a[1].timestamp || 0)
                - Number(b[1].timestamp || 0);
        })
        .slice(-200);

    entries.forEach(([key, item]) => {

        const rawTimestamp = Number(item.timestamp || key || 0);

        const date = rawTimestamp > 100000000000
            ? new Date(rawTimestamp)
            : new Date(rawTimestamp * 1000);

        const dateText = isNaN(date.getTime())
            ? "Unknown"
            : date.toLocaleString();

        // Table
        const row = document.createElement("tr");

        const cells = [
            dateText,
            `${item.temperature ?? ""} °C`,
            `${item.humidity ?? ""} %`,
            item.ldr ?? "",
            item.bulb ? "ON" : "OFF",
            item.mode ?? ""
        ];

        cells.forEach((value) => {
            const cell = document.createElement("td");
            cell.textContent = value;
            row.appendChild(cell);
        });

        table.appendChild(row);

        // Graph
        timestamps.push(
            isNaN(date.getTime())
                ? String(rawTimestamp)
                : date.toLocaleTimeString()
        );

        temperatures.push(Number(item.temperature ?? 0));
        humidities.push(Number(item.humidity ?? 0));
        ldrValues.push(Number(item.ldr ?? 0));

        // CSV
        historyForCSV.push({
            timestamp: dateText,
            temperature: item.temperature ?? "",
            humidity: item.humidity ?? "",
            ldr: item.ldr ?? "",
            bulb: item.bulb ? "ON" : "OFF",
            mode: item.mode ?? ""
        });
    });

    updateChart(
        timestamps,
        temperatures,
        humidities,
        ldrValues
    );
}

        item.temperature,
        item.humidity,
        item.ldr,
        item.bulb,
        item.mode

    const csv = [headers, ...rows]
        .map(row =>
            row.map(value =>
                `"${String(value).replaceAll('"', '""')}"`
            ).join(",")
        )
        .join("\n");

    const blob = new Blob(
        [csv],
        { type: "text/csv;charset=utf-8;" }
    );

    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");

    link.href = url;
    link.download = "smartyy-room-history.csv";

    document.body.appendChild(link);
    link.click();
    link.remove();

    URL.revokeObjectURL(url);
});

Problems Encountered & Fixed Solutions

// LAB TROUBLESHOOTING LOG
⚠️

1. Inverted Relay Switching (Active-LOW Relay)

Issue: The 5V relay module was turning OFF when receiving a HIGH signal and turning ON when receiving a LOW signal, causing the bulb to work backwards.

Fixed Solution: Inverted the relay logic in the code so sending LOW turns the light ON and sending HIGH turns it OFF.
⚠️

2. Incorrect Timestamps Before Wi-Fi & NTP Sync

Issue: When the ESP32 started up before connecting to Wi-Fi, time-series log entries were being saved with 0 or 1970 timestamps.

Fixed Solution: Added a quick check in the firmware loop to verify NTP time is synced before saving any log entries to the cloud.

System Limitations & Future Improvements

// ENGINEERING EVALUATION

Limitations of Current Prototype

  • Sensor Precision: DHT11 sampling rate is limited to 1 Hz with ±2°C temperature and ±5% RH tolerance, precluding clinical or cleanroom applications.
  • Relative Lux Metric: The analog LDR photocell outputs a relative 12-bit ADC value (0–4095) rather than calibrated SI Lux (lumens/m²).
  • Network Single-Point-of-Failure: Device requires active 2.4 GHz Wi-Fi connectivity for cloud synchronization; historical logging pauses during broadband outages.

Possible Future Improvements

  • Precision Sensor Upgrades: Transition to digital I2C sensors like the BME280 (±0.5°C, ±3% RH, barometric pressure) and BH1750 (calibrated digital Lux).
  • Local SPIFFS/SD Card Offline Logging: Implement local flash/SD card buffering to store telemetry during internet outages and batch-sync on reconnection.
  • Edge Machine Learning (TinyML): Deploy an onboard TensorFlow Lite Micro model on ESP32 to predict room occupancy and automate lighting without cloud latency.

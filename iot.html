// ARDUINO C++ SOURCE & SYSTEM TELEMETRY
task1esp32.ino 
#include <WiFi.h>
#include <WebServer.h>

const char* ssid     = "protosem";
const char* password = "Proto#123";

#define LED_PIN 2

WebServer server(80);

void handleRoot()
{
  String html = R"rawliteral(
<!DOCTYPE html>
<html>
<head>
  <title>ESP32 LED Control</title>
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <style>
    body { font-family: Arial; text-align: center; margin-top: 80px; }
    h1 { font-size: 35px; }
    button { width: 150px; height: 60px; font-size: 22px; margin: 15px; cursor: pointer; }
  </style>
</head>
<body>
  <h1>ESP32 LED Control</h1>
  <a href="/on"><button>ON</button></a>
  <a href="/off"><button>OFF</button></a>
</body>
</html>
)rawliteral";

  server.send(200, "text/html", html);
}

void handleLEDOn()
{
  digitalWrite(LED_PIN, HIGH);
  server.send(200, "text/html", "<h1>LED is ON</h1><a href='/'>Back</a>");
}

void handleLEDOff()
{
  digitalWrite(LED_PIN, LOW);
  server.send(200, "text/html", "<h1>LED is OFF</h1><a href='/'>Back</a>");
}

void setup()
{
  Serial.begin(115200);

  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);

  // Connect to Wi-Fi
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");

  while (WiFi.status() != WL_CONNECTED)
  {
    delay(500);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("WiFi connected!");

  Serial.print("ESP32 IP address: ");
  Serial.println(WiFi.localIP());

  // Web server routes
  server.on("/", handleRoot);
  server.on("/on", handleLEDOn);
  server.on("/off", handleLEDOff);

  server.begin();
  Serial.println("Web server started!");
}

void loop()
{
  server.handleClient();
}

// ARDUINO C++ SOURCE & SYSTEM TELEMETRY
esp32_adafruit_mqtt.ino 
#include <WiFi.h>
#include <Adafruit_MQTT.h>
#include <Adafruit_MQTT_Client.h>

const char* ssid     = "protosem";
const char* password = "Proto#123";

#define AIO_SERVER      "io.adafruit.com"
#define AIO_SERVERPORT  1883
#define AIO_USERNAME    "night07owl"
#define AIO_KEY         "AIO_SECRET_KEY"

WiFiClient client;
Adafruit_MQTT_Client mqtt(&client, AIO_SERVER, AIO_SERVERPORT, AIO_USERNAME, AIO_KEY);
Adafruit_MQTT_Publish tempFeed = Adafruit_MQTT_Publish(&mqtt, "night07owl/feeds/temp");

void MQTT_connect() {
  int8_t ret;
  if (mqtt.connected()) return;
  Serial.print("Connecting to MQTT... ");
  while ((ret = mqtt.connect()) != 0) {
    Serial.println(mqtt.connectErrorString(ret));
    Serial.println("Retrying in 5 seconds...");
    mqtt.disconnect();
    delay(5000);
  }
  Serial.println("MQTT Connected!");
}

void setup() {
  Serial.begin(115200);
  analogSetPinAttenuation(34, ADC_11db);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); }
  Serial.println("WiFi connected!");
}

void loop() {
  MQTT_connect();
  float sensorVal = analogRead(34) * (3.3 / 4095.0) * 10.0;
  Serial.print("Publishing temperature: ");
  Serial.println(sensorVal);
  tempFeed.publish(sensorVal);
  mqtt.processPackets(1000);
  mqtt.ping();
  delay(5000);
}

⚠️

Challenge 1 — Periodic 60-Second Socket Dropping

The broker terminated connections whenever sensor reading intervals exceeded 60 seconds. Checking the MQTT specification revealed the keep-alive timeout requirement, which was resolved by integrating non-blocking mqtt.ping() heartbeats.

Using ADC2 pins caused analog reads to return garbage values whenever Wi-Fi was transmitting. Researching the ESP32 silicon errata confirmed that the Wi-Fi driver shares ADC2 SAR; switching the sensor to dedicated ADC1 (GPIO 34) restored clean readings.

esp32_ifttt_webhook.ino 
#include <WiFi.h>
#include <HTTPClient.h>

const char* ssid     = "protosem";
const char* password = "Proto#123";

const char* iftttUrl = "https://maker.ifttt.com/trigger/temp_alert/with/key/A1B2C3D4E5_SECRET_KEY";
unsigned long lastTrigger = 0;
const unsigned long cooldownMs = 120000; // 120-second spam protection lock

void sendIftttAlert(float temp) {
  if (millis() - lastTrigger < cooldownMs) {
    Serial.println("[COOLDOWN] Trigger suppressed to protect API rate limits.");
    return;
  }
  
  HTTPClient http;
  http.begin(iftttUrl);
  http.addHeader("Content-Type", "application/json");
  String payload = "{\"value1\":\"" + String(temp, 1) + " C\",\"value2\":\"Forge Station 03 Overheat\"}";
  int httpCode = http.POST(payload);
  
  if (httpCode > 0) {
    Serial.print("Webhook Sent! HTTP Status: ");
    Serial.println(httpCode);
  }
  http.end();
  lastTrigger = millis();
}

void setup() {
  Serial.begin(115200);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); }
  Serial.println("WiFi connected!");
}

void loop() {
  float currentTemp = analogRead(35) * (3.3 / 4095.0) * 15.0;
  if (currentTemp > 35.0) {
    sendIftttAlert(currentTemp);
  }
  delay(2000);
}

#include <WiFi.h>
#include <FirebaseESP32.h>
#include <DHT.h>

const char* ssid     = "protosem";
const char* password = "Proto#123";

#define FIREBASE_HOST "https://forge-iot-telemetry-default-rtdb.firebaseio.com"
#define FIREBASE_AUTH "FIREBASE_DATABASE_SECRET"
#define DHTPIN 4
#define DHTTYPE DHT11
#define RELAY_PIN 23

DHT dht(DHTPIN, DHTTYPE);
FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

void setup() {
  Serial.begin(115200);
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW);
  dht.begin();
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); }
  
  config.host = FIREBASE_HOST;
  config.signer.tokens.legacy_token = FIREBASE_AUTH;
  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
  Serial.println("Firebase Connected!");
}

void loop() {
  float h = dht.readHumidity();
  float t = dht.readTemperature();
  
  if (!isnan(h) && !isnan(t)) {
    Firebase.setFloat(fbdo, "/forge_station_04/telemetry/humidity_pct", h);
    Firebase.setFloat(fbdo, "/forge_station_04/telemetry/temperature_c", t);
    Serial.printf("[SYNC] Temp: %.1f C, Humidity: %.1f %%\n", t, h);
  }
  
  // Read remote dashboard relay command
  if (Firebase.getBool(fbdo, "/forge_station_04/actuators/relay_powered")) {
    bool state = fbdo.boolData();
    digitalWrite(RELAY_PIN, state ? HIGH : LOW);
  }
  
  delay(3000);
}

PAIN POINTS & FAILURES
  • Single-Threshold Relay Chatter & Mechanical Failure: When a system toggles based on a single threshold (e.g. Light < 1500), sensor noise causes rapid oscillation between 1499 and 1501, making mechanical relays switch 10+ times per second, generating destructive electrical arcs.
  • Cloud Dependency Single Point of Failure: Standard IoT architectures offload automation logic to cloud brokers; when internet connectivity drops or cloud APIs experience latency, local actuators fail to respond to critical environmental changes.
  • Telemetry Data Lock-In & Inaccessible Audits: Microcontroller memory and cloud JSON trees cannot be easily exported for offline analysis by engineers using spreadsheet tools (Excel, pandas, MATLAB) without heavy backend server code.

  • The ESP32 continuously samples an LDR photoresistor on GPIO 34 through internal 12-bit ADC conversion alongside a calibrated DHT11 climate sensor on GPIO 4.
  • An embedded software hysteresis algorithm executes locally at the edge: the 5V relay energizes only when light plunges below 1200 ADC and de-energizes only when light rises above 1500 ADC, completely ignoring the 300-point intermediate zone.
  • Real-time operating metrics (mode, light level, temperature, humidity, bulb status) stream continuously to Google Firebase Realtime Database over encrypted WebSockets.
  • The web dashboard incorporates an instant client-side RFC 4180 CSV export generator that serializes recorded telemetry in browser memory into downloadable spreadsheet files on demand.

Core Firmware & Code Breakdown

// ESP32 EMBEDDED C++ SOURCE
ESP32_Capstone_Automation_Logging.ino 
#include <WiFi.h>
#include <FirebaseESP32.h>
#include <DHT.h>
#include "time.h"

#define WIFI_SSID     "YOUR_WIFI_SSID"
#define WIFI_PASS     "YOUR_WIFI_PASSWORD"
#define FIREBASE_HOST "forge-iot-telemetry-default-rtdb.firebaseio.com"
#define FIREBASE_AUTH "YOUR_DATABASE_SECRET"

#define DHTPIN 4
#define LDRPIN 34
#define RELAYPIN 18

// Software Hysteresis Thresholds (300-point anti-chatter deadband)
const int THRESHOLD_DARK  = 1200;
const int THRESHOLD_LIGHT = 1500;

DHT dht(DHTPIN, DHT11);
FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

unsigned long lastLogTime = 0;
bool currentBulbState = false;
String currentMode = "AUTOMATIC";

void setup() {
  Serial.begin(115200);
  pinMode(RELAYPIN, OUTPUT);
  digitalWrite(RELAYPIN, LOW);
  dht.begin();

  WiFi.begin(WIFI_SSID, WIFI_PASS);
  while (WiFi.status() != WL_CONNECTED) { delay(500); }
  
  configTime(19800, 0, "pool.ntp.org"); // UTC + 5:30 (IST)

  config.host = FIREBASE_HOST;
  config.signer.tokens.legacy_token = FIREBASE_AUTH;
  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
}

void loop() {
  float t = dht.readTemperature();
  float h = dht.readHumidity();
  int ldr = analogRead(LDRPIN);

  // 1. Retrieve Current Mode from Firebase
  if (Firebase.getString(fbdo, "/iot_system/system_state/mode")) {
    currentMode = fbdo.stringData();
  }

  // 2. Execute Hysteresis Control Logic
  if (currentMode == "AUTOMATIC") {
    if (ldr < THRESHOLD_DARK && !currentBulbState) {
      currentBulbState = true;
      digitalWrite(RELAYPIN, HIGH);
      Firebase.setBool(fbdo, "/iot_system/system_state/bulb_state", true);
    } else if (ldr > THRESHOLD_LIGHT && currentBulbState) {
      currentBulbState = false;
      digitalWrite(RELAYPIN, LOW);
      Firebase.setBool(fbdo, "/iot_system/system_state/bulb_state", false);
    }
  } else { // MANUAL OVERRIDE MODE
    if (Firebase.getBool(fbdo, "/iot_system/system_state/bulb_state")) {
      currentBulbState = fbdo.boolData();
      digitalWrite(RELAYPIN, currentBulbState ? HIGH : LOW);
    }
  }

  // 3. Periodic Historical Logging (Every 30 seconds)
  if (millis() - lastLogTime > 30000) {
    lastLogTime = millis();
    FirebaseJson json;
    json.set("temperature", t);
    json.set("humidity", h);
    json.set("light_level", ldr);
    json.set("bulb_state", currentBulbState ? "ON" : "OFF");
    json.set("mode", currentMode);
    
    Firebase.pushJSON(fbdo, "/iot_system/historical_logs", json);
  }
}

Code Breakdown

Architectural examination of the hysteresis decision engine and data logging pipeline

STEP-BY-STEP ANALYSIS
PART 01 THRESHOLD_DARK / LIGHT
1. Dual Threshold Constants
  • Defines lower limit (1200) and upper limit (1500) creating an intentional 300-point deadband.
  • Prevents false switching when environmental shadows or ADC flicker oscillate near setpoints.
PART 02 analogRead(LDRPIN)
2. 12-Bit ADC Sensor Sampling
  • Reads voltage divider on GPIO 34 across 0 to 4095 range with non-blocking execution.
  • Captures physical room lux without introducing execution delays into the control loop.
PART 03 ldr < 1200 && !state
3. Hysteresis Decision Engine
  • Toggles relay ON only when plunging below 1200, and OFF only when climbing past 1500.
  • Maintains current relay state whenever sensor reading is between 1200 and 1500.
PART 04 digitalWrite(RELAYPIN)
4. Optoisolated Actuator Drive
  • Drives GPIO 18 to switch the optocoupled transistor controlling the 5V relay coil.
  • Isolates the 3.3V microcontroller logic safely from inductive coil spikes and AC loads.
PART 05 Firebase.pushJSON()
5. Time-Series Historical Logging
  • Appends immutable timestamped snapshots under /historical_logs every 30 seconds.
  • Feeds the client-side CSV generator with clean, structured historical records for auditing.

Challenges Faced

// TROUBLESHOOTING LOG
⚠️

Challenge 1 — Relay Rapid Oscillations (Chatter) Around Single Threshold

When room light hovered right at a static threshold (1500), subtle atmospheric shadows and sensor noise caused the ADC reading to bounce above and below 1500 multiple times a second, switching the relay noisily and stressing contacts. This was solved by programming dual asymmetric hysteresis limits (1200 / 1500 ADC) creating a solid 300-point deadband.

⚠️

Challenge 2 — Browser Memory Leak During High-Frequency CSV Serialization

Accumulating unbounded telemetry records in browser memory for real-time charting caused DOM sluggishness and excessive RAM usage during long test runs. We resolved this by implementing an in-memory circular buffer capped at 500 records and using lightweight client Blob objects with immediate URL revocation.
